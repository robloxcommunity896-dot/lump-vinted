const STORAGE_KEY = "lump_pro_v2_items";
const GOAL_KEY = "lump_pro_v2_goal";
const THEME_KEY = "lump_pro_v2_theme";
const LOGIN_KEY = "lump_pro_v2_logged";

let items = loadJSON(STORAGE_KEY, []);
let goal = Number(localStorage.getItem(GOAL_KEY) || 0);

function loadJSON(key, fallback){
    try{
        return JSON.parse(localStorage.getItem(key)) ?? fallback;
    }catch{
        return fallback;
    }
}

function save(){
    localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
    localStorage.setItem(GOAL_KEY, String(goal));
}

function money(value){
    return Number(value || 0).toLocaleString("pl-PL",{
        minimumFractionDigits:2,
        maximumFractionDigits:2
    }) + " zł";
}

function integer(value){
    return Number(value || 0).toLocaleString("pl-PL");
}

function pct(value){
    return Number(value || 0).toFixed(1) + "%";
}

function escapeHTML(value){
    return String(value ?? "")
        .replaceAll("&","&amp;")
        .replaceAll("<","&lt;")
        .replaceAll(">","&gt;")
        .replaceAll('"',"&quot;")
        .replaceAll("'","&#039;");
}

    }

    document

    document

        .classList.remove("hidden");

    updateAll();
}

/* =========================================================
   ITEM CALCULATOR
========================================================= */

function calculatePreview(){

    const qty =
        Number(document.getElementById("itemQuantity")?.value) || 0;

    const buy =
        Number(document.getElementById("itemBuy")?.value) || 0;

    const sell =
        Number(document.getElementById("itemSell")?.value) || 0;

    const costs =
        Number(document.getElementById("itemCosts")?.value) || 0;

    const totalCost =
        qty * buy + costs;

    const totalRevenue =
        qty * sell;

    const totalProfit =
        totalRevenue - totalCost;

    const profitPerItem =
        qty > 0
            ? totalProfit / qty
            : 0;

    const roi =
        totalCost > 0
            ? totalProfit / totalCost * 100
            : 0;

    document.getElementById("previewProfit")
        .textContent = money(profitPerItem);

    document.getElementById("previewROI")
        .textContent = pct(roi);

    document.getElementById("previewCost")
        .textContent = money(totalCost);

    document.getElementById("previewTotalProfit")
        .textContent = money(totalProfit);
}

function addItem(){

    const brand =
        document.getElementById("itemBrand")
        .value.trim();

    const category =
        document.getElementById("itemCategory").value;

    const quantity =
        Number(document.getElementById("itemQuantity").value) || 1;

    const buy =

    const sell =
        Number(document.getElementById("itemSell").value) || 0;

    const costs =
        Number(document.getElementById("itemCosts").value) || 0;

    const minProfit =
        Number(document.getElementById("itemMinProfit").value) || 0;

    const image =
        document.getElementById("itemImage").value.trim();

    const note =
        document.getElementById("itemNote").value.trim();

    if(!brand){
        toast("Podaj markę produktu.","⚠");
        return;
    }

    if(buy <= 0 || sell <= 0){
        toast("Podaj prawidłową cenę zakupu i sprzedaży.","⚠");
        return;
    }

    const item = {

        id:
            Date.now() +
            Math.random(),

        brand,

        quantity,
        buy,
        sell,
        costs,
        minProfit,
        image,

        status:"bought",

        listedAt:null,
        soldAt:null,
        soldPrice:null

    items.unshift(item);

    save();

    clearAddForm();

    toast("Produkt został dodany do inventory.","💎");

    showPage("inventory");
}

/* =========================================================

function itemCost(item){
    return item.quantity * item.buy + item.costs;
}

function itemPotentialRevenue(item){
    return item.quantity * item.sell;
}

function itemPotentialProfit(item){
    return itemPotentialRevenue(item) - itemCost(item);
}

function itemRealProfit(item){
    return Number(item.soldPrice || 0) - itemCost(item);
}

/* =========================================================
   INVENTORY
========================================================= */

function renderInventory(){

    const grid =
        document.getElementById("itemsGrid");

    if(!grid) return;

    const search =
        (
            document.getElementById("inventorySearch")
                ?.value || ""
        ).toLowerCase();

    const category =
        document.getElementById("inventoryCategory")
            ?.value || "";

    const status =
        document.getElementById("inventoryStatus")
            ?.value || "";

    const sort =
        document.getElementById("inventorySort")
            ?.value || "newest";

    let filtered =
        items.filter(item=>{

            const text =
                `${item.brand} ${item.category} ${item.note}`
                    .toLowerCase();

            return (
                text.includes(search) &&
                (!category || item.category===category) &&
                (!status || item.status===status)
            );

        });

    if(sort==="profit"){

        filtered.sort(
            (a,b)=>
                itemPotentialProfit(b) -
                itemPotentialProfit(a)
        );

    }else if(sort==="buyLow"){

        filtered.sort(
            (a,b)=>a.buy-b.buy
        );

    }else if(sort==="sellHigh"){

        filtered.sort(
            (a,b)=>b.sell-a.sell
        );

    }else{

        filtered.sort(
            (a,b)=>
                new Date(b.createdAt) -
                new Date(a.createdAt)
        );

    }

    if(!filtered.length){

        grid.innerHTML = emptyHTML(
            "📦",
            "Brak produktów",
            "Dodaj pierwszy produkt do inventory."
        );

        return;
    }

    grid.innerHTML =
        filtered.map(itemCard).join("");
}

/* =========================================================
   STATS ENGINE
========================================================= */

function calculateStats(){

    const spent =
        items.reduce(
            (sum,x)=>sum+itemCost(x),
            0
        );

    const active =
        items.filter(
            x=>x.status!=="sold"
        );

    const inventoryValue =
        active.reduce(
            (sum,x)=>sum+itemCost(x),
            0
        );

    const potentialRevenue =
        active.reduce(
            (sum,x)=>sum+itemPotentialRevenue(x),
            0
        );

    const potentialProfit =
        active.reduce(
            (sum,x)=>sum+itemPotentialProfit(x),
            0
        );

    const sold =
        items.filter(
            x=>x.status==="sold"
        );

    const revenue =
        sold.reduce(
            (sum,x)=>sum+Number(x.soldPrice||0),
            0
        );

    const realProfit =
        sold.reduce(
            (sum,x)=>sum+itemRealProfit(x),
            0
        );

    const roi =
        spent > 0
            ? realProfit / spent * 100
            : 0;

    const avgProfit =
        sold.length
            ? realProfit / sold.length
            : 0;

    const soldThrough =
        items.length
            ? sold.length / items.length * 100
            : 0;

    return {
        spent,
        inventoryValue,
        potentialRevenue,
        potentialProfit,
        soldRevenue:revenue,
        realProfit,
        roi,
        sold,
        avgProfit,
        soldThrough
    };
}

/* =========================================================
   CHART
========================================================= */

function renderChart(){

    const chart =
        document.getElementById("chartBars");

    if(!chart) return;

    const now = new Date();

    const days = [];

    for(let i=13;i>=0;i--){

        const date =
            new Date(now);

        date.setHours(0,0,0,0);
        date.setDate(
            date.getDate()-i
        );

        const value =
            items
                .filter(
                    x=>
                        x.status==="sold" &&
                        x.soldAt &&
                        new Date(x.soldAt)
                            .toDateString() ===
                        date.toDateString()
                )
                .reduce(
                    (sum,x)=>
                        sum+itemRealProfit(x),
                    0
                );

        days.push({
            date,
            value
        });
    }

    const max =
        Math.max(
            ...days.map(x=>Math.max(0,x.value)),
            1
        );

    const total =
        days.reduce(
            (sum,x)=>sum+x.value,
            0
        );

    document.getElementById("chartTotal")
        .textContent = money(total);

    chart.innerHTML =
        days.map(day=>{

            const height =
                Math.max(
                    3,
                    Math.max(0,day.value) /
                    max *
                    100
                );

            return `

                <div class="chart-bar-wrap">

                    <div
                        class="chart-bar"
                        title="${money(day.value)}"
                        style="height:${height}%">
                    </div>

                    <div class="chart-label">
                        ${day.date.getDate()}
                    </div>

                </div>

            `;

        }).join("");
}

function soldProfitThisMonth(){

    const now =
        new Date();

    return items
        .filter(
            x=>
                x.status==="sold" &&
                x.soldAt
        )
        .filter(x=>{

            const date =
                new Date(x.soldAt);

            return (
                date.getMonth() === now.getMonth() &&
                date.getFullYear() === now.getFullYear()
            );

        })
        .reduce(
            (sum,x)=>
                sum+itemRealProfit(x),
            0
        );
}

                    margin-top:14px">
                    +${money(itemRealProfit(best))}
                </div>

            </div>
        `;
    }

    const now =
        new Date();

    const today =
        items
            .filter(
                x=>
                    x.status==="sold" &&
                    x.soldAt
            )
            .filter(
                x=>
                    new Date(x.soldAt)
                        .toDateString() ===
                    now.toDateString()
            )
            .reduce(
                (sum,x)=>
                    sum+itemRealProfit(x),
                0
            );

    const year =
        items
            .filter(
                x=>
                    x.status==="sold" &&
                    x.soldAt &&
                    new Date(x.soldAt)
                        .getFullYear() ===
                    now.getFullYear()
            )
            .reduce(
                (sum,x)=>
                    sum+itemRealProfit(x),
                0
            );

    document.getElementById("summaryToday")
        .textContent = money(today);

    document.getElementById("summaryMonth")
        .textContent = money(
            soldProfitThisMonth()
        );

    document.getElementById("summaryYear")
        .textContent = money(year);

    document.getElementById("summaryAll")
        .textContent = money(s.realProfit);
}

function setTheme(a,b){

    document.documentElement
        .style.setProperty("--accent",a);

    document.documentElement
        .style.setProperty("--accent-2",b);

    localStorage.setItem(
        THEME_KEY,
        JSON.stringify({a,b})
    );

    toast(
        "Motyw został zmieniony.",
        "🎨"
    );
}

function loadTheme(){

    const saved =
        loadJSON(THEME_KEY,null);

    if(!saved) return;

    document.documentElement
        .style.setProperty(
            "--accent",
            saved.a
        );

    document.documentElement
        .style.setProperty(
            "--accent-2",
            saved.b
        );
}

   DATA EXPORT
========================================================= */

function exportData(){

    const data = {

        app:"LUMP PRO",
        version:2,

        exportedAt:
            new Date().toISOString(),

        items,
        goal

    };

    const blob =
        new Blob(
            [JSON.stringify(data,null,2)],
            {type:"application/json"}
        );

    const url =
        URL.createObjectURL(blob);

    const a =
        document.createElement("a");

    a.href=url;
    a.download=
        `lump-pro-backup-${new Date()
            .toISOString()
            .slice(0,10)}.json`;

    a.click();

    URL.revokeObjectURL(url);

    toast(
        "Backup został pobrany.",
        "↓"
    );
}

function importData(event){

    const file =
        event.target.files[0];

    if(!file) return;

    const reader =
        new FileReader();

    reader.onload = ()=>{

        try{

            const data =
                JSON.parse(reader.result);

            if(!Array.isArray(data.items)){
                throw new Error();
            }

            items =
                data.items;

            goal =
                Number(data.goal || 0);

            save();
            updateAll();

            closeModal();

            toast(
                "Dane zostały zaimportowane.",
                "↑"
            );

        }catch{

            toast(
                "Nieprawidłowy plik backupu.",
                "❌"
            );
        }
    };

    reader.readAsText(file);
}

/* =========================================================
   KEYBOARD
========================================================= */

document.addEventListener("keydown",event=>{

    if(
        (event.ctrlKey || event.metaKey) &&
        event.key.toLowerCase()==="k"
    ){

        event.preventDefault();
        openCommand();

    }

    if(event.key==="Escape"){

        closeCommand();
        closeModal();

    }

    if(
        event.altKey &&
        event.key>="1" &&
        event.key<="5"
    ){

        const pages=[
            "dashboard",
            "add",
            "inventory",
            "sold",
            "stats"
        ];

        const page =
            pages[Number(event.key)-1];

        if(page){
            showPage(page);
        }
    }
});
